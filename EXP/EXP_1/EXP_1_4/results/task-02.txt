CREATE VIEW V_SSG AS
SELECT s.SNO,s.SName,g.GoodsName,g.SalePrice,b.Number
FROM Student AS s JOIN SaleBill AS b ON b.SNO=s.SNO
JOIN Goods AS g ON g.GoodsNO=b.GoodsNO;
EXIT: 0
