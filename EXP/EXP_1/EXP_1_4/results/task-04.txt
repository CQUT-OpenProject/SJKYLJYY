CREATE VIEW V_COM AS SELECT College,COUNT(*) AS StudentCount,
CAST(AVG(YEAR(CURRENT_DATE())-BirthYear) AS DECIMAL(6,2)) AS AvgAge
FROM Student GROUP BY College;
