SQL:
SELECT g.GoodsNO,g.GoodsName,s.SName,b.HappenTime,b.Number
FROM SaleBill AS b JOIN Goods AS g ON g.GoodsNO=b.GoodsNO
JOIN Student AS s ON s.SNO=b.SNO
ORDER BY b.HappenTime DESC,g.GoodsNO,s.SNO;
